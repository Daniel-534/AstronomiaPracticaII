\centering
\begin{tikzpicture}[>=Stealth,scale=0.94]
  \coordinate (O) at (-7.5,2.4);
  \coordinate (V) at (0,0);
  \coordinate (F) at (-2,0);
  \coordinate (C) at (-4,0);
  \coordinate (I) at (-2.727,-0.873);

  % Eje óptico y espejo cóncavo abierto hacia la izquierda.
  \draw[->] (-8.2,0) -- (1.15,0) node[right] {eje óptico};
  \draw[very thick,blue!70!black,smooth,domain=-3:3,samples=60]
    plot ({0.12*(1-\x*\x/9)},{\x});
  \foreach \y in {-2.6,-1.3,0,1.3,2.6}{
    \pgfmathsetmacro{\xm}{0.12*(1-\y*\y/9)}
    \draw[gray!65] (\xm,\y) -- ({\xm+0.27},{\y+0.17});
  }
  \node[align=center] at (0.72,2.55) {espejo\\cóncavo};

  % Puntos notables.
  \fill (V) circle (1.8pt) node[above right=3pt] {$V$};
  \fill (F) circle (1.8pt) node[above=4pt] {$F$};
  \fill (C) circle (1.8pt) node[above=4pt] {$C$};

  % 1. Rayo paralelo al eje que se refleja pasando por F.
  \draw[blue,thick,->] (O) -- (0,2.4);
  \draw[blue,thick,->] (0,2.4) -- (-4,-2.4);

  % 2. Rayo que pasa por F y se refleja paralelo al eje.
  \draw[green!55!black,thick,->] (O) -- (F) -- (0,-0.873);
  \draw[green!55!black,thick,->] (0,-0.873) -- (-5.8,-0.873);

  % 3. Rayo dirigido hacia C: incide normalmente y regresa sobre sí mismo.
  \draw[purple!80!black,thick,->] (O) -- (C) -- (0,-2.743);
  \draw[purple!80!black,thick,dashed,->] (0,-2.743) -- (-6,1.371);

  % 4. Rayo que incide en V y se refleja simétricamente respecto al eje.
  \draw[orange!90!black,thick,->] (O) -- (V);
  \draw[orange!90!black,thick,->] (V) -- (-5,-1.6);

  % Objeto e imagen común determinada por la intersección de los rayos.
  \draw[->,very thick] (-7.5,0) -- (O) node[above] {objeto};
  \draw[->,very thick,red!70!black] (-2.727,0) -- (I)
    node[below=3pt] {imagen};

  % Distancias características.
  \draw[dashed,gray!55] (-7.5,0) -- (-7.5,-3.65);
  \draw[dashed,gray!55] (-2.727,0) -- (-2.727,-3.18);
  \draw[dashed,gray!55] (0,0) -- (0,-3.65);
  \draw[<->] (-7.5,-3.55) -- (0,-3.55)
    node[midway,fill=white,inner sep=2pt] {$d_o$};
  \draw[<->] (-2.727,-3.05) -- (0,-3.05)
    node[midway,fill=white,inner sep=2pt] {$d_i$};

  % Leyenda de los cuatro rayos principales.
  \draw[blue,thick,->] (-7.4,-4.15) -- (-6.75,-4.15);
  \node[right,font=\small] at (-6.65,-4.15) {1. paralelo $\rightarrow F$};
  \draw[green!55!black,thick,->] (-3.2,-4.15) -- (-2.55,-4.15);
  \node[right,font=\small] at (-2.45,-4.15) {2. por $F$ $\rightarrow$ paralelo};
  \draw[purple!80!black,thick,dashed,->] (-7.4,-4.65) -- (-6.75,-4.65);
  \node[right,font=\small] at (-6.65,-4.65) {3. por $C$ $\rightarrow$ retorna};
  \draw[orange!90!black,thick,->] (-3.2,-4.65) -- (-2.55,-4.65);
  \node[right,font=\small] at (-2.45,-4.65) {4. en $V$ $\rightarrow$ simétrico};
\end{tikzpicture}
\caption{Formación de una imagen real, invertida y reducida mediante los cuatro rayos principales de un espejo cóncavo.}
\end{figure}
